# charla-vista: the charla pair with the roles swapped (A watches by the fire, B talks), so the harness captures of A show what the watcher sees.
# charla, B: follows the same way as A (crawl-space tunnel, then east of the root) and stops a few
# metres from the Kokiri camp fire facing it, to watch A talk. B is the owner of the camp's NPCs when
# it lists them first, so A's talk must claim them (LICHII_COOP_NPC=1).
mark start
wait 30
goto -76.3 13.4 1.2 1500     # the crawl-space lintel
toward -73.0 13.4 10
wait 10
press A 12                   # "Enter"
wait 120
stick 0 1 600                # through the tunnel
mark out-tunnel
wait 60
goto -73 -12 1.5 1500
goto -73.5 -16 1.5 900       # east of the fire
toward -78.9 -17 10          # face the camp
mark watching
wait 1500
mark end
