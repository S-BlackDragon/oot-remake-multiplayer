#! noload
#! start coop: test warp done
#! switches LICHII_COOP_PUPPET=1,LICHII_COOP_DEBUG_FLAGS=1,LICHII_COOP_DUNGEON_LISTS=shared_dungeon_minigames.txt
#! expect A coop: lists: .*shared_dungeon_minigames\.txt: 30 flags and keys added, 0 taken off, 28 once, 0 held in their area, 0 one-time keys within boot..
#! expect A coop: test flag SubEvent_FindRichard_GetHeartPiece \(0xf2fab510\) = 1: queued
#! expect A coop: test flag Npc_SkullKid_Child_IsGetHeartPiece \(0x5ff7aa30\) = 1: queued
#! expect A coop: test flag Minigame_SkullKidSession_IsGetHeartPiece \(0x1657d7d8\) = 1: queued
mark start
warp Market_Child
wait 1200               # the title screen
mark title
press A
wait 180
press A
wait 180
press A
wait 300
press A                 # loads the save, then the warp
wait 2400               # the save loads, then the warp lands (about 30 s)
mark after
wait 600                # the other game lands too: puppets made
mark control
flag SubEvent_FindRichard_GetHeartPiece 1          # a prize given: one for the group
flag Npc_SkullKid_Child_IsGetHeartPiece 1          # a prize given: one for the group
flag Minigame_SkullKidSession_IsGetHeartPiece 1          # a prize given: one for the group
wait 600
wait 300
