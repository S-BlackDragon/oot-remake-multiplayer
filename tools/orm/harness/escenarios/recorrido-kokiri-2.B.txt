#! start coop: test warp done

#! timeout 900
#! expect B coop: test warp done
#! never B coop: test goto timed out
#! never B coop: test place .*: failed
#! expect B coop: puppet: P\d:? (create )?CoopPuppet_Child within llegada..r01-entrada
#! never B (?<!fairy )coop: puppet: P\d show -> hide within r01-entrada..salida
#! expect B coop: lists: .*shared_dungeon_child\.txt: \d+ flags and keys added within boot..
#! expect B coop: test place 1\.20 100\.34 4\.97: done within r01-entrada..r02-cruce
#! expect B coop: test place 25\.20 100\.38 \-29\.15: done within r02-cruce..r03-pradera
#! expect B coop: flags debug: \w+ 0xa61c116d \(entry \d+\) 0 -> 1 within r02-cruce..fin
#! never B coop: flags debug: \w+ 0xa61c116d \(entry \d+\) 1 -> 0
#! expect B coop: test place 25\.20 100\.34 \-71\.13: done within r03-pradera..salida
#! expect B coop: drops: asked for the drops of SacredForestMeadow within salida..fin
#! expect B coop: puppet: P\d:? (create )?CoopPuppet_Child within salida..fin
mark start
warp LostWood_Child LostWood_Entrance
wait 1200               # the title screen
mark title
press A
wait 180
press A
wait 180
press A
wait 300
press A                 # loads the save, then the warp
wait 2400               # the save loads, then the warp lands (about 30 s)
mark llegada
wait 600                # the other game is in already, or lands now
mark r01-entrada            # la entrada desde el pueblo
place 1.20 100.34 4.97
wait 120
goto 1.20 -2.00 1.5 900
wait 300                # the other one sees us in this room
mark r02-cruce            # el cruce del lado Goron
place 25.20 100.38 -29.15
wait 120
goto 15.70 -29.00 1.5 900
wait 300                # the other one sees us in this room
mark r03-pradera            # ante la salida a la Pradera
place 25.20 100.34 -71.13
wait 120
goto 25.20 -77.13 1.5 900
wait 300                # the other one sees us in this room
mark salida             # a la Pradera
goto 24.60 -83.70 0.8 900
wait 1800               # the scene change, then the other one lands too
mark fin
