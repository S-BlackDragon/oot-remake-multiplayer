# recorrido-kakariko-1 (ORM, docs/RECORRIDOS.md, bloque 6. Kakariko y Cementerio), B: tramo 1, Kakariko hasta el Cementerio.
# KakarikoVillage_Child: warp to KakarikoVillage_Entrance, then room by room with place/goto (B 1.2 m beside A); see recorrido-kakariko.steps.txt.
# Run alone: r03_pareja.py --scenario recorrido-kakariko-1 --transport ws (switches from the header).
#! noload
#! start coop: test warp done
#! saves 06-kakariko
#! switches LICHII_COOP_PUPPET=1,LICHII_COOP_DEBUG_OBJECTS=1,LICHII_COOP_DEBUG_FLAGS=1,LICHII_COOP_DEBUG_BOSS=1,LICHII_COOP_STORY=1,LICHII_COOP_DUNGEON_LISTS=shared_dungeon_child.txt
#! timeout 900
#! expect B coop: test warp done
#! never B coop: test goto timed out
#! never B coop: test place .*: failed
#! expect B coop: puppet: P\d:? (create )?CoopPuppet_Child within llegada..r01-entrada
#! never B (?<!fairy )coop: puppet: P\d show -> hide within r01-entrada..salida
#! expect B coop: lists: .*shared_dungeon_child\.txt: \d+ flags and keys added within boot..
#! expect B coop: test place \-118\.57 96\.93 58\.70: done within r01-entrada..r02-molino
#! expect B coop: test place \-8\.91 104\.93 43\.34: done within r02-molino..r03-puerta-montana
#! expect B coop: flags debug: \w+ 0xc7fe3216 \(entry \d+\) 0 -> 1 within r02-molino..fin
#! never B coop: flags debug: \w+ 0xc7fe3216 \(entry \d+\) 1 -> 0
#! expect B coop: test place \-40\.00 110\.29 \-10\.00: done within r03-puerta-montana..r04-cementerio
#! expect B coop: test place \-4\.53 101\.84 68\.80: done within r04-cementerio..salida
#! never B coop: flags debug: \w+ 0xf8823cf0 \(entry \d+\) 0 -> 1 within r03-puerta-montana..salida
#! expect B coop: flags debug: \w+ 0xf8823cf0 \(entry \d+\) 0 -> 1 within salida..fin
#! expect B coop: drops: asked for the drops of Graveyard within salida..fin
#! expect B coop: puppet: P\d:? (create )?CoopPuppet_Child within salida..fin
mark start
warp KakarikoVillage_Child KakarikoVillage_Entrance
wait 1200               # the title screen
mark title
press A
wait 180
press A
wait 180
press A
wait 300
press A                 # loads the save, then the warp
wait 2400               # the save loads, then the warp lands (about 30 s)
mark llegada
wait 600                # the other game is in already, or lands now
mark r01-entrada            # la entrada desde el campo
place -118.57 96.93 58.70
wait 120
goto -103.80 55.00 1.5 900
wait 300                # the other one sees us in this room
mark r02-molino            # el molino
place -8.91 104.93 43.34
wait 120
goto -18.80 40.00 1.5 900
wait 300                # the other one sees us in this room
mark r03-puerta-montana            # la puerta de la Montaña (guardia, carta de Zelda)
place -40.00 110.29 -10.00
wait 120
goto -40.00 -10.00 1.5 900
wait 300                # the other one sees us in this room
mark r04-cementerio            # el camino del Cementerio
place -4.53 101.84 68.80
wait 120
goto 6.20 66.00 1.5 900
wait 300                # the other one sees us in this room
mark salida             # al Cementerio
goto 17.32 64.14 0.8 900
wait 1800               # the scene change, then the other one lands too
mark fin
